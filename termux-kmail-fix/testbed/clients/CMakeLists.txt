cmake_minimum_required(VERSION 3.20)
project(walletclients CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 REQUIRED COMPONENTS Core Gui)
find_package(KF6Wallet REQUIRED)
find_package(Qt6Keychain REQUIRED)
add_executable(async_client async_client.cpp)
target_link_libraries(async_client Qt6::Gui KF6::Wallet)
add_executable(keychain_client keychain_client.cpp)
target_link_libraries(keychain_client Qt6::Gui qt6keychain)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
add_executable(window_client window_client.cpp)
target_link_libraries(window_client Qt6::Widgets KF6::Wallet)
